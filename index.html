import { AI, DevManagement, Notion, Minpaku, Antique } from '@starkid/core';
// ── 会社概要 ─────────────────────────────────────
const company = {
name: "St.Arkid",
founded: 2020,
location: "東京都",
concept: "テクノロジーと人の暮らしをつなぐ",
services: [
"AI活用支援", // 🤖 AI consulting
"開発プロジェクトマネジメント", // 🛠 Dev management
"Notionアンバサダー", // 📋 Notion ambassador
"民泊事業（Airbnb）", // 🏠 Minpaku
"古物商", // 🛒 Antique dealer
]
};
// ── AI活用支援 ────────────────────────────────────
@Service("AI")
class AIConsulting {

describe() {

/**

* St.ArkidのAI活用支援サービス

*

* ChatGPT・Claude・Geminiなどの生成AIを

* ビジネスに組み込むための導入コンサルティングから

* 業務自動化・プロンプト設計まで一貫支援。

* AI × 業務改善で、チームの生産性を最大化します。

*/

return {

skills: ["生成AI導入", "プロンプトエンジニアリング", "業務自動化", "RAG構築"],

};

}
}
// ── 開発プロジェクトマネジメント ─────────────────────
@Service("DevManagement")
class DevelopmentManagement {

describe() {

/**

* 開発プロジェクトマネジメント

*

* Webアプリ・モバイルアプリ・社内ツール開発の

* プロジェクトマネジメントを担います。

* スクラム・アジャイル手法によるチームビルディング、

* 要件定義から納品・運用フェーズまでを一貫してサポート。

*/

return {

methodology: ["Scrum", "Agile", "Waterfall"],

tools: ["Jira", "GitHub", "Notion", "Figma"],

scope: ["要件定義", "設計", "開発管理", "QA", "リリース"]

};

}
}
// ── Notionアンバサダー ────────────────────────────
@Service("NotionAmbassador")
class NotionAmbassador {

describe() {

/**

* Notion公認アンバサダーとして

* 個人・チーム・企業向けNotionコンサルを提供。

*

* データベース設計・社内Wiki構築・業務フロー整備など、

* Notionを最大限に活用できる環境をゼロから構築します。

* Notion AI活用支援も対応可能。

*/

return {

certified: true,

services: ["DB設計", "社内Wiki構築", "Notion AI活用", "研修・セミナー"]

};

}
}

*

* 物件オーナー向けの民泊代行・収益最大化コンサルも承ります。

services: ["物件管理", "ゲスト対応", "収益コンサル", "運用代行"]

};

}
}
